function updateDailyRewardsUI(user){
  const streak = user.loginStreak || 0;
  const lastClaimStr = user.lastDailyReward;

  // Calculate if claim is ready
  let isClaimReady = true;
  if(lastClaimStr){
    const diffMs = Date.now() - new Date(lastClaimStr);
    const hoursPassed = diffMs / (1000 * 60 * 60);
    if(hoursPassed < 24){
      isClaimReady = false;
    }
  }

  document.getElementById("streakCounterText").textContent = streak + " Day Streak";
  // calculate % based on 7 days cycle
  const currentProgressPercent = Math.min(100, Math.round(((streak % 7) || (streak > 0 ? 7 : 0)) / 7 * 100));
  document.getElementById("streakProgressBar").style.width = currentProgressPercent + "%";

  const claimBtn = document.getElementById("claimRewardBtn");
  if(isClaimReady){
    claimBtn.disabled = false;
    claimBtn.innerHTML = '<i class="fas fa-gift"></i> Claim Today\'s Reward';
    claimBtn.style.background = "var(--gradient-main)";
    claimBtn.style.boxShadow = "var(--shadow-glow)";
  } else {
    claimBtn.disabled = true;
    claimBtn.innerHTML = '<i class="fas fa-lock"></i> Claimed (Locked)';
    claimBtn.style.background = "#222";
    claimBtn.style.boxShadow = "none";
  }

  // Format day cards
  // activeDay is next claim day. If streak is 3 and claimed, next is day 4. If streak is 3 and unclaimed, it's day 3.
  let currentStreakDay = (streak % 7);
  if (currentStreakDay === 0 && streak > 0) currentStreakDay = 7;

  for(let day=1; day<=7; day++){
    const card = document.getElementById("streakDay" + day);
    if(!card) continue;

    if(day < currentStreakDay){
      // Already claimed in current cycle
      card.style.background = "rgba(0, 255, 136, 0.05)";
      card.style.border = "1px solid rgba(0, 255, 136, 0.3)";
      card.style.opacity = "0.6";
      card.querySelector("i").className = "fas fa-check-circle";
      card.querySelector("i").style.color = "#00ff88";
    } else if(day === currentStreakDay && !isClaimReady) {
      // Claimed today
      card.style.background = "rgba(0, 255, 136, 0.05)";
      card.style.border = "1px solid rgba(0, 255, 136, 0.3)";
      card.style.opacity = "0.6";
      card.querySelector("i").className = "fas fa-check-circle";
      card.querySelector("i").style.color = "#00ff88";
    } else if(day === currentStreakDay && isClaimReady) {
      // Ready to claim
      card.style.background = "rgba(255, 0, 60, 0.1)";
      card.style.border = "2px solid var(--primary-red)";
      card.style.opacity = "1";
      card.style.transform = "scale(1.05)";
      card.style.boxShadow = "0 0 15px rgba(255, 0, 60, 0.2)";
      card.querySelector("i").className = "fas fa-gift";
      card.querySelector("i").style.color = "gold";
    } else {
      // Locked for future days
      card.style.background = "rgba(255, 255, 255, 0.01)";
      card.style.border = "1px solid rgba(255, 255, 255, 0.05)";
      card.style.opacity = "0.4";
      card.style.transform = "none";
      card.style.boxShadow = "none";
      if (day === 7) {
        card.querySelector("i").className = "fas fa-gift";
        card.querySelector("i").style.color = "gold";
      } else {
        card.querySelector("i").className = "fas fa-lock";
        card.querySelector("i").style.color = "rgba(255,255,255,0.4)";
      }
    }
  }
}

async function purchaseAndDownloadAsset(id){
  const token=getToken();
  if(!token){
    alert("Please register or log in first to purchase and download assets!");
    showSection("authView");
    return;
  }

  try{
    const r=await fetch(API+`/api/assets/${id}/download`,{
      method:"POST",
      headers:{
        "Content-Type":"application/json",
        Authorization:"Bearer "+token
      }
    });
    const data=await r.json();
    if(!r.ok){
      alert(data.error || "Failed to initiate download.");
      return;
    }

    // If successfully bought/unlocked, update user's coin balance locally in the header
    if(data.userCoins !== undefined){
      document.getElementById("dashCoins").textContent=data.userCoins;
      const info=JSON.parse(localStorage.getItem("userInfo")||"{}");
      info.coins=data.userCoins;
      localStorage.setItem("userInfo",JSON.stringify(info));
    }

    // Trigger the file download in browser
    const link=document.createElement("a");
    link.href=data.downloadUrl;
    // Extract file name
    link.download=data.downloadUrl.split("/").pop();
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);

    alert("Download started successfully!");
  }catch(e){
    alert("An error occurred during asset download checkout.");
  }
}

async function loadAdminIncomeStats(){
  const tbody = document.getElementById("incomeTableBody");
  const secContainer = document.getElementById("sectionViewsContainer");

  try {
    const r = await fetch(API+"/admin/ads/analytics", {
      headers: { Authorization: "Bearer " + getToken() }
    });
    const data = await r.json();
    if(!r.ok){
      tbody.innerHTML = "<tr><td colspan=5 style='color:#ff003c; text-align:center;'>Failed to retrieve ad insights.</td></tr>";
      return;
    }

    const ads = data.ads || [];

    let totalViews = 0;
    let totalClicks = 0;
    let totalUsd = 0;

    const adRows = [];

    ads.forEach(a => {
      totalViews += a.views || 0;
      totalClicks += a.clicks || 0;
      totalUsd += a.earnings || 0;

      adRows.push(`
        <tr style="border-bottom:1px solid var(--border-color)">
          <td style="padding:10px; font-weight:600; color:#fff;">${a.adUnit}</td>
          <td style="padding:10px; color:var(--text-secondary);">${a.adId}</td>
          <td style="padding:10px; color:#00ff88; font-weight:600;">${(a.views || 0).toLocaleString()}</td>
          <td style="padding:10px;">${a.clicks || 0} (${a.ctr || "0%"})</td>
          <td style="padding:10px; color:gold; font-weight:600;">$${(a.earnings || 0).toFixed(2)}</td>
        </tr>
      `);
    });

    tbody.innerHTML = adRows.join("") || "<tr><td colspan=5 style='text-align:center;'>No AdSense units registered.</td></tr>";

    // Update main dashboard metrics
    document.getElementById("adEstIncomeUsd").textContent = "$" + totalUsd.toFixed(2);
    // INR conversion (assuming 1 USD = 83 INR)
    document.getElementById("adEstIncomeInr").textContent = "₹" + (totalUsd * 83).toLocaleString(undefined, {minimumFractionDigits:2, maximumFractionDigits:2}) + " INR";
    document.getElementById("adTotalViews").textContent = totalViews.toLocaleString();
    document.getElementById("adTotalHours").textContent = totalClicks.toLocaleString();

    // Render static section breakdown for clean AdSense layout
    const mockedSections = [
      { name: "dashboard", views: Math.round(totalViews * 0.55) },
      { name: "mods", views: Math.round(totalViews * 0.25) },
      { name: "skins", views: Math.round(totalViews * 0.12) },
      { name: "daily-rewards", views: Math.round(totalViews * 0.08) }
    ];

    const maxSecViews = mockedSections[0].views || 1;

    secContainer.innerHTML = mockedSections.map(s => {
      const pct = Math.min(100, Math.round((s.views / maxSecViews) * 100));
      return `
        <div>
          <div style="display:flex; justify-content:space-between; font-size:12px; margin-bottom:4px;">
            <span style="text-transform:capitalize; font-weight:600; color:#fff;">${s.name.replace("-", " ")} Section</span>
            <span style="color:gold; font-weight:700;">${s.views.toLocaleString()} views</span>
          </div>
          <div style="width:100%; height:6px; background:rgba(255,255,255,0.05); border-radius:10px; overflow:hidden;">
            <div style="width:${pct}%; height:100%; background:linear-gradient(90deg, #ff003c, gold); border-radius:10px;"></div>
          </div>
        </div>
      `;
    }).join("");

  } catch(e) {
    console.error(e);
    tbody.innerHTML = "<tr><td colspan=5 style='color:#ff003c; text-align:center;'>Network error connecting to AdSense servers.</td></tr>";
  }
}

    const ramPercent = Math.min(100, Math.round((usedMb / totalMb) * 100)) || 15;
    document.getElementById("memoryBar").style.width = ramPercent + "%";

    // Live CPU fluctuation simulation
    const cpuVal = (Math.random() * 8 + 1).toFixed(1);
    document.getElementById("cpuUsage").textContent = cpuVal + " %";
    document.getElementById("cpuBar").style.width = cpuVal + "%";

    const ar=await fetch(API+"/admin/activity",{headers:{Authorization:"Bearer "+getToken()}});
    if(ar.ok){const ad=await ar.json();document.getElementById("recentActivity").innerHTML=(ad.activity||[]).slice(0,5).map(i=>`<div style="padding:8px 0;border-bottom:1px solid var(--border-color); font-size:12px;"><i class="fas fa-caret-right" style="color:var(--primary-red); margin-right:6px;"></i>${i.details}</div>`).join("")||"No activity"}

    // Render Top Countries
    const topCountries = s.topCountries || [];
    const countryEl = document.getElementById("adminTopCountries");
    if (countryEl) {
      if (!topCountries.length) {
        countryEl.innerHTML = "<p style='color:var(--text-secondary); font-size:12px;'>No country statistics recorded yet.</p>";
      } else {
        const maxCount = topCountries[0].count || 1;
        countryEl.innerHTML = topCountries.map(c => {
          const pct = Math.min(100, Math.round((c.count / maxCount) * 100));
          return `
            <div>
              <div style="display:flex; justify-content:space-between; font-size:12px; margin-bottom:4px;">
                <span style="font-weight:600; color:#fff;">${c._id}</span>
                <span style="color:gold; font-weight:700;">${c.count.toLocaleString()} user(s)</span>
              </div>
              <div style="width:100%; height:6px; background:rgba(255,255,255,0.05); border-radius:10px; overflow:hidden;">
                <div style="width:${pct}%; height:100%; background:linear-gradient(90deg, #00ff88, #00aa55); border-radius:10px;"></div>
              </div>
            </div>
          `;
        }).join("");
      }
    }

    // Render Gender/Pronoun Demographics
    const genderDemographics = s.genderDemographics || [];
    const genderEl = document.getElementById("adminGenderDistribution");
    if (genderEl) {
      if (!genderDemographics.length) {
        genderEl.innerHTML = "<p style='color:var(--text-secondary); font-size:12px;'>No gender statistics recorded yet.</p>";
      } else {
        const totalGenderUsers = genderDemographics.reduce((acc, curr) => acc + curr.count, 0) || 1;
        genderEl.innerHTML = genderDemographics.map(g => {
          const pct = Math.min(100, Math.round((g.count / totalGenderUsers) * 100));
          let label = g._id;
          if (g._id === "he/him") label = "Male (he/him)";
          if (g._id === "she/her") label = "Female (she/her)";
          if (g._id === "they/them") label = "Non-binary (they/them)";
          if (g._id === "other") label = "Other / Prefer not to say";
          return `
            <div>
              <div style="display:flex; justify-content:space-between; font-size:12px; margin-bottom:4px;">
                <span style="font-weight:600; color:#fff; text-transform:capitalize;">${label}</span>
                <span style="color:var(--primary-red); font-weight:700;">${pct}% (${g.count})</span>
              </div>
              <div style="width:100%; height:6px; background:rgba(255,255,255,0.05); border-radius:10px; overflow:hidden;">
                <div style="width:${pct}%; height:100%; background:linear-gradient(90deg, var(--primary-red), #8b0000); border-radius:10px;"></div>
              </div>
            </div>
          `;
        }).join("");
      }
    }

  }catch(e){console.error(e)}
}

async function initializePublicSettings(){
  try {
    const r = await fetch(API+"/api/settings");
    if(r.ok){
      const data = await r.json();
      const s = data.settings || {};
      if(s.serverName){
        document.title = s.serverName + " - Epic Gaming Platform";
        document.getElementById("logoTextSpan").textContent = s.serverName;
        document.getElementById("welcomeTitleSpan").textContent = "Welcome to " + s.serverName;
      }
    }
  } catch(e) {
    console.warn("Could not load public settings", e);
  }
}

document.getElementById("saveOnboardingBtn").onclick = async () => {
  const country = document.getElementById("onboardingCountry").value;
  const pronouns = document.getElementById("onboardingPronouns").value;
  const msg = document.getElementById("onboardingMsg");

  if (!country || !pronouns) {
    return showMsg(msg, "Please select both your country and pronouns.", false);
  }

  try {
    const r = await fetch(API + "/api/profile/onboarding", {
      method: "POST",
      headers: {
        "Content-Type": "application/json",
        Authorization: "Bearer " + getToken()
      },
      body: JSON.stringify({ country, pronouns })
    });
    const data = await r.json();
    if (!r.ok) return showMsg(msg, data.error || "Failed to save details", false);

    showMsg(msg, "Profile setup successful!", true);
    // Update local storage info
    const info = JSON.parse(localStorage.getItem("userInfo") || "{}");
    Object.assign(info, data.profile || {});
    localStorage.setItem("userInfo", JSON.stringify(info));
    currentUser = info;

    setTimeout(() => {
      document.getElementById("onboardingModal").classList.remove("active");
      loadDashboard();
    }, 800);
  } catch (e) {
    showMsg(msg, "Network error", false);
  }
};
